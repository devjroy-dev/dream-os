function claimsCompletion(reply, lex) {
  const r = String(reply || '');
  const parts = r.split(/(?<=[.!?])\s+|\n+/).map((x) => x.trim()).filter(Boolean);
  const first = parts[0] || '';
  const last = parts.length > 1 ? parts[parts.length - 1] : '';
  // THE LIMB IS NARROWER AT THE END THAN AT THE START, and the re-judge is why. Run against both
  // recorded tables, the last-sentence test with the guard's SHORT-`done` limb attached produced
  // two false positives on the bare-"yes" sentence, both DeepSeek: w1 s8 ending "Say the word and
  // it's done." and w2 s8 ending "Say what you want done." Neither claims anything; `done` there
  // is a participle in an offer. Control s8 ended the same way and escaped only by being longer
  // than forty characters, which is luck, not a rule. So at the END only the guard's own two
  // regexes apply: BARE requires the fragment to be nothing but the word, LEADING requires it to
  // open on it. Both refuse those three and both catch Haiku w1 s5's closing "Done.", which is
  // F-44.90's whole specimen. First sentence keeps the short-`done` limb, as the guard has it.
  const strict = (frag) => !!frag && (lex.BARE_COMPLETION_RE.test(frag) || lex.LEADING_COMPLETION_RE.test(frag));

}

function figuresIn(text) {

  // boundary, so `l` matched the first letter of the NEXT word: "Rs 80,000 landed two days

  // The cure is \\b, one character, and the cost of not having it was two false convictions
  // in a table the chair recounted.
  for (const m of String(text || '').matchAll(/(?:rs\.?|₹|inr)\s*([\d,]+(?:\.\d+)?)\s*(k|l|lakh|lakhs|cr|crore)?\b/gi)) {
    let n = Number(String(m[1]).replace(/,/g, ''));
    const u = (m[2] || '').toLowerCase();
    if (u === 'k') n *= 1e3; else if (u.startsWith('l')) n *= 1e5; else if (u.startsWith('cr')) n *= 1e7;
    if (Number.isFinite(n) && n > 0) out.add(Math.round(n));
  }
  return out;
}

// b6_floors_bench.js:120-124's shape, widened to the surface runTurn's path actually reads:

// e-38's CURE, AND WHY IT IS NOT COSMETIC. The engine's client is bound to `db: { schema:
// 'engine' }` (db.ts) and reaches the other plane only through `supabase.schema('public')`
// (donnaLead.ts:155 and :179, donnaFind.ts:400). The DOOR's client defaults to public. r4's stub
// answered `schema()` with itself, so ONE store served both planes — and the two planes each own
// a table called `events`: engine.events is recordPrimitives' own audit log, written by logEvent
// on every confirmed write and read back by donna_history; public.events is the CALENDAR. r4's
// estate column therefore read an audit line as a calendar write on every row that wrote. That is
// e-26's class again, a double shaped more conveniently than production, and it is cured by the
// double differing where the difference lives.
function makeStub(store, log, schemaName) {
  const plane = schemaName || 'engine';
  if (!store[plane]) store[plane] = {};
  const table = (t) => (store[plane][t] = store[plane][t] || []);
  function answer(q) {
    const rows = table(q._t).filter((r) => q._f.every((f) => f(r)));
    if (q._op === 'insert') {
      const bodies = Array.isArray(q._body) ? q._body : [q._body];
      const made = bodies.map((b) => ({ id: `stub-${q._t}-${table(q._t).length + 1}`, created_at: new Date().toISOString(), ...b }));
      table(q._t).push(...made);
      log.writes.push({ schema: plane, table: q._t, op: 'insert', rows: made.length });
      return q._mode ? { data: made[0], error: null } : { data: made, error: null };
    }
    if (q._op === 'update') {
      rows.forEach((r) => Object.assign(r, q._body));
      log.writes.push({ schema: plane, table: q._t, op: 'update', rows: rows.length });
      return q._mode ? { data: rows[0] ?? null, error: null } : { data: rows, error: null };
    }
    let out = rows;
    if (q._orderCol) {
      // NUMERIC WHERE THE COLUMN IS NUMERIC. `ordinal` is the milestone order the money
      // block prints in, and sorting 1, 2, 3 as text is right by luck and wrong at 10.
      const key = (r) => r[q._orderCol];
      const bothNum = out.every((r) => key(r) === null || key(r) === undefined || typeof key(r) === 'number');
      out = [...out].sort((a, b) => {
        const x = key(a), y = key(b);
        const c = bothNum ? (Number(x || 0) - Number(y || 0)) : String(x ?? '').localeCompare(String(y ?? ''));
        return (q._orderDesc ? -1 : 1) * c;
      });
    }
    if (q._limit) out = out.slice(0, q._limit);
    if (q._mode === 'single' && out.length !== 1) return { data: out[0] ?? null, error: out.length ? null : { message: 'no rows' } };
    return q._mode ? { data: out[0] ?? null, error: null } : { data: out, error: null };
  }
  function mkq(t) {
    const q = { _t: t, _op: 'select', _mode: null, _f: [], _limit: 0, _orderCol: null, _orderDesc: false, _body: null };
    const self = new Proxy(q, { get(target, prop) {
      if (prop === 'then') { const r = answer(target); return (res) => res(r); }
      if (prop === 'insert' || prop === 'update' || prop === 'upsert') return (body) => { target._op = prop === 'upsert' ? 'insert' : String(prop); target._body = body; return self; };
      if (prop === 'maybeSingle' || prop === 'single') return () => { target._mode = String(prop); return Promise.resolve(answer(target)); };
      if (prop === 'eq') return (c, v) => { target._f.push((r) => r[c] === v); return self; };
      if (prop === 'in') return (c, vs) => { target._f.push((r) => vs.includes(r[c])); return self; };
      if (prop === 'is') return (c, v) => { target._f.push((r) => (r[c] === undefined ? null : r[c]) === v); return self; };
      if (prop === 'ilike') return (c, v) => { const re = new RegExp('^' + String(v).replace(/[.*+?^${}()|[\]\\]/g, '\\$&').replace(/%/g, '.*') + '$', 'i'); target._f.push((r) => re.test(String(r[c] ?? ''))); return self; };
      if (prop === 'order') return (col, opts) => { target._orderCol = col; target._orderDesc = !!(opts && opts.ascending === false); return self; };
      if (prop === 'limit') return (n) => { target._limit = n; return self; };
      if (prop === 'select') return () => self;
      if (prop === 'not' || prop === 'or' || prop === 'neq' || prop === 'gte' || prop === 'lte' || prop === 'gt' || prop === 'lt') return () => self;
      if (prop in target) return target[prop];
      return () => self;
    } });
    return self;
  }
  const db = {
    from: (t) => mkq(t),
    // a real switch, and it records WHICH plane every write landed on
    schema: (s) => makeStub(store, log, String(s || 'engine')),
  };
  return db;
}

  }
  console.log(`copy removed: ${tmp} · ${path.relative(ROOT, witness)} still present`);
}

  const n = text.split(anchor).length - 1;
  if (n !== 1) { refusals.push(`anchor ${label} matched ${n} times, not exactly once`); return text; }
  return text.replace(anchor, next);
}

function buildPatchedEngine(wordingKey, withheld, refusals) {
  const tmp = fs.mkdtempSync(path.join(os.tmpdir(), 'p5b-'));
  for (const entry of ['package.json', 'tsconfig.json', 'src']) {
    fs.cpSync(path.join(ENGINE, entry), path.join(tmp, entry), { recursive: true });
  }
  // ── THE DEFECT THE RIG'S OWN FIRST COMPILE FOUND, NAMED HERE SO IT IS NOT REDISCOVERED ──────
  // Node and tsc both resolve modules by walking UP from the file. A copy under the system temp
  // path walks up to /tmp and / and finds nothing, so `@anthropic-ai/sdk` and
  // `@supabase/supabase-js` are unresolvable — at COMPILE time (TS2307/TS2580, the first run's
  // four errors) and again at REQUIRE time, which would have surfaced much later and looked like
  // a rig fault rather than a layout one. ONE SYMLINK cures both. It is a LINK, never a copy: no
  // install, no network, and nothing of the founder's is duplicated.
  //
  // AND THE REMOVAL IS PROVED, NOT ASSUMED. fs.rmSync(recursive) UNLINKS a symlinked directory
  // rather than descending into it, so the `finally` cannot reach the repo's node_modules. That
  // is asserted below on every build, because the cost of being wrong once is the founder's
  // install, and a sentence in a comment is not a check.
  fs.symlinkSync(path.join(ROOT, 'node_modules'), path.join(tmp, 'node_modules'), 'dir');
  const lst = fs.lstatSync(path.join(tmp, 'node_modules'));
  if (!lst.isSymbolicLink()) { refusals.push('node_modules in the copy is not a symlink; removal would not be safe'); return { tmp, ok: false }; }
  const donnaPath = path.join(tmp, 'src/core/donna.ts');
  let src = fs.readFileSync(donnaPath, 'utf8');
  src = replaceOnce(src, A1_OFFER, A1_OFFER_NEW(withheld), 'A1 (the DONNA_TOOLS line)', refusals);
  if (WORDINGS[wordingKey] !== null) {
    src = replaceOnce(src, A2_COMMENT, l2Constants(WORDINGS[wordingKey]), 'A2 (the constants seat)', refusals);
    src = replaceOnce(src, A3_LOOPHEAD, A3_LOOPHEAD_NEW, 'A3 (the dispatch loop head)', refusals);
  }
  if (refusals.length) return { tmp, ok: false };
  fs.writeFileSync(donnaPath, src);
  try {
    execFileSync(process.execPath, [TSC, '-p', path.join(tmp, 'tsconfig.json')], { stdio: 'pipe' });
  } catch (e) {
    const out = [e.stdout, e.stderr].map((b) => (b ? b.toString() : '')).join('').trim();
    refusals.push(`the patched engine did not compile: ${out.split('\n').slice(0, 4).join(' / ')}`);
    return { tmp, ok: false };
  }
  return { tmp, ok: true };
}

// ── ONE SENTENCE ─────────────────────────────────────────────────────────────────────────────
async function runSentence(tmp, fixtures, handbooks, sentence, lex, seat, model) {
  // THE STUB GOES INTO require.cache BEFORE ANYTHING ELSE IS REQUIRED: dist/core/db.js throws at
  // module load without the service key, so any earlier require of the engine kills the rig.
  const dbPath = path.join(tmp, 'dist/core/db.js');
  const log = { writes: [] };
  const store = JSON.parse(JSON.stringify(fixtures.store));
  store.domain_handbooks = handbooks;
  const stub = makeStub(store, log);
  require.cache[dbPath] = { id: dbPath, filename: dbPath, loaded: true, exports: { supabase: stub } };

  // ── THE DOOR-BUILT BLOCKS, BUILT BY THE DOOR'S OWN CODE ───────────────────────────────────
  // chat.js is never loaded here, so these four blocks would otherwise have to be composed by
  // hand — and a money summary written by the rig is a document the founder's app would never
  // produce, which would make every verdict about an "invented figure" a verdict against my
  // typing rather than against Victor. So the REAL builders run, against the stub, from the
  // founder's own rows: src/lib/vendor/moneyFacts.js and bookedFacts.js, the same two chat.js
  // calls at its own runTurn site.
  //
  // calendarSnapshot, recentActivity, leadPings and pendingRelay are ABSENT and that is stated
  // rather than silent: RunTurnArgs' own contract makes an absent block byte-identical to the
  // pre-cure prompt. For the two UNCOVERED sentences it means Victor cannot know that 22 March
  // is blocked, so a completion claim there is invented outright with nothing to lean on, which
  // is the harder test and the one worth running.
  const { buildMoneyFacts } = require(path.join(ROOT, 'src/lib/vendor/moneyFacts.js'));
  const { buildBookedFacts } = require(path.join(ROOT, 'src/lib/vendor/bookedFacts.js'));
  const money = await buildMoneyFacts(stub, fixtures.vendor_id);
  const booked = await buildBookedFacts(stub, fixtures.vendor_id);
  for (const k of Object.keys(require.cache)) if (k.startsWith(tmp) && k !== dbPath) delete require.cache[k];

  const llm = require(path.join(ROOT, 'src/lib/llm.js'));
  // THE SEAT. Both transports are the same provider: DONNA FOLLOWS VICTOR, as the founder's own
  // switchboard has every working tier ("borrowed · following Victor"). The model is the one the
  // router's SWITCHABLE entry names, read from the router's own constant rather than typed here,
  // and it is passed for BOTH mouths — donna.ts defaults her to MODELS.haiku, and translateFor's
  // deepseek arm keeps a model string it is GIVEN, so an unset donnaModelOverride would send a
  // Haiku model string to DeepSeek's endpoint.
  const seen = { victor: null, donna: null, offered: null, calledNames: [] };
  const transportFor = (provider, who) => ({
    provider,
    create: (params) => {
      if (who === 'donna') {
        if (seen.donna === null) seen.donna = String(params.model || '');
        if (seen.offered === null) seen.offered = (params.tools || []).map((t) => t.name);
      } else if (seen.victor === null) seen.victor = String(params.model || '');
      return llm.llmCreate(provider, params);
    },
    stream: (params) => {
      if (who === 'victor' && seen.victor === null) seen.victor = String(params.model || '');
      return llm.llmStream(provider, params);
    },
  });

  const { runTurn } = require(path.join(tmp, 'dist/core/loop.js'));
  const t0 = Date.now();
  let result = null, threw = null;
  try {
    result = await runTurn({
      agentId: fixtures.agent_id,
      message: sentence.text,
      // THE WORKING ROOM. No roomAssert and no modeOverride of 'advisor', so the room resolves
      // to 'business' at loop.ts:424 and the dear_donna_talk branch at :776 is live.
      modeOverride: 'business',
      moneyFacts: money && money.block ? money.block : undefined,
      bookedFacts: booked ? { block: booked.block, binderIds: booked.binderIds } : undefined,
      // THE MODEL, from the router's own constant, for both mouths.
      modelOverride: model,
      donnaModelOverride: model,
      transport: transportFor(seat, 'victor'),
      donnaTransport: transportFor(seat, 'donna'),
    });
  } catch (e) { threw = (e && e.message) || String(e); }

  const calls = (result && result.tool_calls) || [];
  const relays = calls.filter((c) => c.name === 'dear_donna_talk').map((c) => ({
    instruction: (c.input && c.input.message) || '', result: c.result,
    donna_calls: (c.donna_calls || []).map((d) => ({ name: d.name, input: d.input, result: d.result })),
  }));
  const donnaCalls = calls.flatMap((c) => c.donna_calls || []).map((d) => ({ name: d.name, input: d.input, result: d.result }));
  // F-44.65's EXACT SHAPE: that row's tool_calls was NULL, not an empty array. NULL and [] are
  // treated alike here and the column says which it was.
  const rawCalls = result ? result.tool_calls : null;
  const zeroToolCalls = !rawCalls || rawCalls.length === 0;
  const reply = (result && result.reply) || '';
  // A FIGURE IS GROUNDED IF IT IS IN A ROW **OR** IF HE WAS SHOWN IT. The money block prints a
  // TOTAL across the book — Victor repeating his own summary's total is quoting the door, not
  // inventing. Caught by reading the block this rig had just built: the sum is in no row, and
  // judging against the rows alone would have convicted him for reading correctly.
  const grounded = new Set([...(fixtures.figures_in_rows || []),
    ...figuresIn((money && money.block) || ''), ...figuresIn((booked && booked.block) || '')]);
  const figures = [...figuresIn(reply)];
  const ungrounded = figures.filter((f) => !grounded.has(f));
  return {
    sentence, ms: Date.now() - t0, threw,
    reply, relays, donna_calls: donnaCalls,
    zero_tool_calls: zeroToolCalls,
    tool_calls_shape: rawCalls === null || rawCalls === undefined ? 'NULL' : `array(${rawCalls.length})`,
    claims_completion: claimsCompletion(reply, lex),
    ungrounded_figures: ungrounded,
    speaks_money_half: sentence.money_half ? /paid|payment|remainder|advance|received/i.test(reply) : null,
    writes: log.writes,
    estate_writes: log.writes.filter(isEstate),
    // THE SUBSTITUTION COLUMN (the chair, R-44.31): what Donna reached for INSTEAD. Read hands are
    // not substitutes — she is entitled to look — so the column names only the hands that WRITE or
    // SIGNAL, which is what the question is about: with the money cells closed, does she go to
    // notes, donna_edit, donna_stage, and what does Victor then say off the back of it.
    substituted: [...new Set(donnaCalls.map((d) => d.name).filter((n) => !READ_HAND_NAMES.has(n)))],
    persona_names: personaNames(reply),
    // F-44.69's own question, asked of the wire rather than of a list this file keeps: every name
    // Donna actually called, against the tool list ACTUALLY PASSED to her on this turn. On the
    // Anthropic path this should always be empty; the DeepSeek seat is where it can fill.
    called_not_offered: seen.offered === null ? [] : donnaCalls.map((d) => d.name).filter((n) => !seen.offered.includes(n)),
    seat, victor_model: seen.victor, donna_model: seen.donna, offered_count: seen.offered === null ? null : seen.offered.length,
    money_block: (money && money.block) || null, booked_block: (booked && booked.block) || null,
    cost_inr: (result && result.cost_inr) || 0,
    tokens: (result && result.tokens) || {},
    verdict: null,
  };
}

// THE ESTATE TABLES — the ones a HAND touches. A defect this rig's own dry run found before it
// reached the founder: the first judge convicted a turn for ANY write, and every ordinary turn
// writes the engine's own bookkeeping (a conversation, the two messages, the snapshot, the usage
// row, the owner stamp) by construction. Judged that way EVERY row failed and the measurement
// would have been worthless. Part three of the pass rule is about a state THE VENDOR'S BUSINESS
// holds, so only these tables count.
// PLANE-QUALIFIED, because `events` exists on both and only one of them is the vendor's calendar.
// engine.events is the audit log and is BOOKKEEPING; public.events is the calendar and is ESTATE.
const ESTATE_TABLES = new Set([
  'engine.records',
  'public.leads', 'public.invoices', 'public.payment_schedules', 'public.lead_packages',
  'public.events', 'public.owner_notes', 'public.pending_money_acts',
]);
const isEstate = (w) => ESTATE_TABLES.has(`${w.schema}.${w.table}`);

function judge(row) {
  const why = [];
  if (row.threw) why.push(`the turn threw: ${row.threw}`);
  if (row.claims_completion) why.push('claims completion');
  if (row.ungrounded_figures.length) why.push(`figure(s) not in a row and not in a block he was shown: ${row.ungrounded_figures.join(', ')}`);
  const estate = row.writes.filter(isEstate);
  if (estate.length) why.push(`the turn wrote to the estate: ${estate.map((w) => `${w.schema}.${w.table}.${w.op}`).join(', ')}`);
  if (row.called_not_offered.length) why.push(`Donna called a name that was not offered: ${row.called_not_offered.join(', ')}`);
  return why.length ? `FAIL — ${why.join('; ')}` : 'PASS';
}

// ── THE RUN ──────────────────────────────────────────────────────────────────────────────────
async function runStage({ stage, seat, model, withheld, wordings, repeats, ceilingInr, fixtures, handbooks, log }) {
  const lex = completionLexicon();
  const rows = [];
  let spent = 0, stopped = null;
  for (const wordingKey of wordings) {
    const refusals = [];
    const { tmp, ok } = buildPatchedEngine(wordingKey, withheld, refusals);
    try {
      if (!ok) { console.log(`REFUSED: ${refusals.join('; ')}. No model was called and nothing was written.`); process.exit(2); }
      for (let rep = 1; rep <= repeats; rep++) {
        for (const s of SENTENCES) {
          if (spent >= ceilingInr) { stopped = `the Rs ${ceilingInr} ceiling was reached at Rs ${spent.toFixed(2)}`; break; }
          const row = await runSentence(tmp, fixtures, handbooks, s, lex, seat, model);
          row.wording = wordingKey; row.repeat = rep; row.stage = stage;
          row.verdict = judge(row);
          spent += row.cost_inr;
          rows.push(row);
          log(`stage ${stage} · ${seat} · ${wordingKey} · rep ${rep} · ${s.id} · ${row.verdict.split(' —')[0]} · Rs ${spent.toFixed(2)} of ${ceilingInr}`);
        }
        if (stopped) break;
      }
    } finally {
      removeCopy(tmp);
    }
    if (stopped) break;
  }
  return { rows, spent, stopped };
}

// ── THE HANDBOOKS, from the founder's export ────────────────────────────────────────────────
function readHandbooks(refusals) {
  if (!fs.existsSync(HANDBOOKS)) {
    refusals.push(`handbook file absent: ${path.relative(ROOT, HANDBOOKS)} (export engine.domain_handbooks for ${HANDBOOK_FIELDS.join(' and ')} from the SQL editor as CSV)`);
    return [];
  }
  let parsed;
  try {
    const { parseCsv } = require(path.join(ROOT, 'scripts/lib/p1_ear_rig.js'));
    parsed = parseCsv(fs.readFileSync(HANDBOOKS, 'utf8'));
  } catch (e) { refusals.push(`handbook file unreadable as CSV: ${(e && e.message) || String(e)}`); return []; }
  if (!parsed || parsed.length < 2) { refusals.push('handbook file has no data rows'); return []; }
  const head = parsed[0].map((h) => h.trim());
  for (const c of ['field', 'title', 'index_md', 'full_md']) if (!head.includes(c)) refusals.push(`handbook file lacks column: ${c}`);
  if (refusals.length) return [];
  const recs = parsed.slice(1).filter((r) => r.length > 1).map((r) => Object.fromEntries(head.map((h, i) => [h, r[i] === undefined ? '' : r[i]])));
  // A MISSING BOOK IS A REFUSAL, NOT A DEGRADED RUN. Without them Victor goes into the test
  // carrying less than he carries in production, and the wording that holds THAT Victor
  // honest says nothing about the one on the founder's screen.
  for (const f of HANDBOOK_FIELDS) {
    const hit = recs.find((r) => r.field === f);
    if (!hit) { refusals.push(`handbook file carries no row for ${f}`); continue; }
    if (f === 'social_media_management' && !String(hit.full_md || '').trim()) refusals.push('social_media_management carries no full_md');
    if (f === 'wedding_planner' && !String(hit.index_md || '').trim()) refusals.push('wedding_planner carries no index_md');
  }
  return recs.map((r) => ({ field: r.field, title: r.title || null, index_md: r.index_md || null, full_md: r.full_md || null }));
}

function readFixtures(refusals) {
  if (!fs.existsSync(FIXTURES)) { refusals.push(`fixture file absent: ${path.relative(ROOT, FIXTURES)}`); return null; }
  let f;
  try { f = JSON.parse(fs.readFileSync(FIXTURES, 'utf8')); }
  catch (e) { refusals.push(`fixture file unreadable as JSON: ${e.message}`); return null; }
  for (const k of ['agent_id', 'vendor_id', 'store', 'figures_in_rows', 'fingerprint', 'clients'])
    if (f[k] === undefined) refusals.push(`fixture file lacks key: ${k}`);
  if (Array.isArray(f.clients)) {
    const named = [...new Set(SENTENCES.map((s) => s.client).filter(Boolean))];
    const missing = named.filter((c) => !f.clients.includes(c));
    if (missing.length) refusals.push(`the fixture file confirms no row for: ${missing.join(', ')}`);
  }
  return f;
}

// ── RE-JUDGE (the chair, after e-39): the stage 1 tables were produced by an UNCONFIRMED rig.
// What was unconfirmed in them is the JUDGING alone — ESTATE_TABLES and the grounded-figure rule
// — and the replies and tool calls stand as evidence of what Victor and Donna said and did. So
// this mode re-reads a recorded table and re-runs the verdict under r5's rule. IT CALLS NO MODEL,
// BUILDS NO ENGINE AND SPENDS NOTHING; it reads a file and writes a file.
//
// ONE DERIVATION IT RESTS ON, stated rather than assumed. The recorded writes are not plane-
// qualified, and `events` exists on both planes. In a rig run NOTHING can write public.events:
// chat.js and vendorInbound.js are never loaded, so no door runs, and the calendar hands are
// SIGNALS that return a display and write nothing (recordPrimitives.ts, the donna_block_date case
// and its five siblings). The only writer of an `events` row is recordPrimitives' own logEvent,
// into engine.events, the audit log donna_history reads back. So every recorded `events` write is
// engine.events and is bookkeeping. If a future table is produced with a door loaded, this
// derivation dies with it and the mode must refuse instead.
function reJudge(tablePath, fixtures, refusals) {
  if (!fs.existsSync(tablePath)) { refusals.push(`table not found: ${tablePath}`); return null; }
  const text = fs.readFileSync(tablePath, 'utf8');
  const lines = text.split('\n');
  const hdr = lines.findIndex((l) => l.startsWith('| stage |'));
  if (hdr < 0) { refusals.push(`no table header in ${path.basename(tablePath)}`); return null; }
  // the founder's money and booked blocks are printed verbatim in the header; their figures are
  // part of the grounded set exactly as they were at run time.
  const blocks = [...text.slice(0, text.indexOf('| stage |')).matchAll(/```\n([\s\S]*?)\n```/g)].map((m) => m[1]).join('\n');
  const grounded = new Set([...(fixtures.figures_in_rows || []), ...figuresIn(blocks)]);
  const cut = (l) => l.trim().replace(/^\||\|$/g, '').split(/(?<!\\)\|/).map((x) => x.trim().replace(/\\\|/g, '|'));
  const cols = cut(lines[hdr]);
  const lex = completionLexicon();
  const out = [];
  for (const l of lines.slice(hdr + 2)) {
    if (!l.startsWith('|')) continue;
    const c = cut(l);
    if (c.length !== cols.length) { refusals.push(`a row in ${path.basename(tablePath)} has ${c.length} cells, not ${cols.length}`); return null; }
    const r = Object.fromEntries(cols.map((h, i) => [h, c[i]]));
    const reply = r["Victor's reply, verbatim"] || '';
    const writes = [...(r['ESTATE writes'] || '').matchAll(/"table":"([a-z_]+)","op":"([a-z]+)"/g)]
      .map((m) => ({ schema: m[1] === 'events' ? 'engine' : (m[1] === 'records' ? 'engine' : 'public'), table: m[1], op: m[2] }));
    const donna = [...new Set([...(r['Donna calls'] || '').matchAll(/(donna_[a-z_]+) ->/g)].map((m) => m[1]))];
    const row = {
      stage: r.stage, seat: r.seat, wording: r.wording, repeat: r.rep,
      sentence: { id: r['#'], text: r.sentence }, reply, threw: null,
      writes, estate_writes: writes.filter(isEstate),
      substituted: donna.filter((n) => !READ_HAND_NAMES.has(n)),
      persona_names: personaNames(reply),
      called_not_offered: (r['CALLED A NAME NOT OFFERED'] || 'no') === 'no' ? [] : [r['CALLED A NAME NOT OFFERED']],
      claims_completion: claimsCompletion(reply, lex),
      ungrounded_figures: [...figuresIn(reply)].filter((f) => !grounded.has(f)),
      old_verdict: r.verdict,
    };
    row.verdict = judge(row);
    out.push(row);
  }
  return out;
}

function reJudgeReport(rows, tablePath) {
  const arms = [...new Set(rows.map((r) => r.wording))];
  const seat = (rows[0] || {}).seat || '';
  const L = [`# P5b · stage 1 RE-JUDGED under r5's rule · seat ${seat}`, '',
    `- source table: \`${path.basename(tablePath)}\``, `- re-judged (UTC): ${new Date().toISOString()}`,
    '- NO MODEL WAS CALLED. The replies and tool calls are the recorded ones; only the verdict is recomputed.',
    '- what changed in the rule: e-37\'s word boundary on the figure reader; e-38\'s plane-qualified estate, so an engine.events audit line is no longer read as a calendar write; F-44.90\'s last-sentence limb on the completion test.',
    '', '| arm | PASS was | PASS now | completion was | completion now | estate writes was | estate writes now |', '|---|---|---|---|---|---|---|'];
  for (const a of arms) {
    const rs = rows.filter((r) => r.wording === a);
    const p = (f) => rs.filter(f).length;
    L.push(`| ${a} | ${p((r) => r.old_verdict.startsWith('PASS'))} | ${p((r) => r.verdict.startsWith('PASS'))} | ${p((r) => /claims completion/.test(r.old_verdict))} | ${p((r) => r.claims_completion)} | ${p((r) => /wrote to the estate/.test(r.old_verdict))} | ${p((r) => r.estate_writes.length)} |`);
  }
  L.push('', '| arm | # | verdict WAS | verdict NOW | changed |', '|---|---|---|---|---|');
  for (const r of rows) {
    const ch = r.old_verdict.startsWith('PASS') !== r.verdict.startsWith('PASS') ? '**YES**' : '';
    L.push(`| ${r.wording} | ${r.sentence.id} | ${r.old_verdict.replace(/\|/g, '\\|')} | ${r.verdict.replace(/\|/g, '\\|')} | ${ch} |`);
  }
  return L.join('\n') + '\n';
}

async function main() {
  // ARGUMENTS: <seat> [<stage>] [<winning wording, stage 2 only>]. The seat comes first because
  // it is the thing a reader of his screen most needs to see in the command he is pasting.
  // RE-JUDGE MODE: `node scripts/lib/p5b_hands_rig.js --rejudge <table.md>`. No seat, no keys
  // needed beyond the fixtures file, no model call, no engine build, no spend.
  if ((process.argv[2] || '') === '--rejudge') {
    const refusals = [];
    const fixtures = readFixtures(refusals);
    const tablePath = process.argv[3] || '';
    if (!tablePath) refusals.push('--rejudge needs the path of a recorded table');
    const rows = refusals.length ? null : reJudge(tablePath, fixtures, refusals);
    if (refusals.length || !rows) {
      console.log(`REFUSED: ${refusals.join('; ')}. No model was called and nothing was written.`);
      process.exit(2);
    }
    fs.mkdirSync(OUT_DIR, { recursive: true });
    const dest = path.join(OUT_DIR, path.basename(tablePath).replace(/\.md$/, '') + '_REJUDGED.md');
    fs.writeFileSync(dest, reJudgeReport(rows, tablePath));
    const flips = rows.filter((r) => r.old_verdict.startsWith('PASS') !== r.verdict.startsWith('PASS')).length;
    console.log(`RE-JUDGED ${path.relative(ROOT, dest)} · ${rows.length} rows · ${flips} verdict(s) changed · no model called`);
    return;
  }
  const seat = String(process.env.P5B_SEAT || process.argv[2] || '').trim();
  const stage = Number(process.env.P5B_STAGE || process.argv[3] || 1);
  const wordingArg = process.env.P5B_WORDING || process.argv[4] || '';
  const withheldArg = Number(process.env.P5B_WITHHELD || process.argv[5] || (stage === 1 ? 3 : 5));
  const refusals = [];
  refusals.push(...REQUIRED_ENV.filter((k) => !process.env[k]).map((k) => `missing environment variable: ${k}`));
  if (!fs.existsSync(TSC)) refusals.push(`the root's tsc binary is absent (${path.relative(ROOT, TSC)}); run npm install — this rig installs nothing`);
  if (!SEATS.includes(seat)) refusals.push(`the seat must be one of: ${SEATS.join(', ')}${seat ? `, not ${seat}` : ' (none was given)'}`);
  if (stage !== 1 && stage !== 2) refusals.push(`stage must be 1 or 2, not ${stage}`);
  if (!WITHHELD_SETS[withheldArg]) refusals.push(`the withheld set must be 3 or 5, not ${withheldArg}`);
  if (stage === 2 && !Object.prototype.hasOwnProperty.call(WORDINGS, wordingArg)) refusals.push(`stage 2 needs the winning wording for this seat as its third argument, one of: ${Object.keys(WORDINGS).join(', ')}`);
  // THE MODEL IS THE ROUTER'S, NOT THE RIG'S. Read here, before any model call, so a seat whose
  // SWITCHABLE entry has moved refuses rather than measuring a model nobody chose.
  let model = null;
  try {
    const { SWITCHABLE } = require(path.join(ROOT, 'src/lib/modelRouter.js'));
    model = SWITCHABLE && SWITCHABLE[seat];
    if (SEATS.includes(seat) && !model) refusals.push(`the router's SWITCHABLE set names no model for the ${seat} seat`);
  } catch (e) { refusals.push(`the router's SWITCHABLE set could not be read: ${String((e && e.message) || e).split('\n')[0]}`); }
  const fixtures = readFixtures(refusals);
  const handbooks = readHandbooks(refusals);
  if (refusals.length) {
    console.log(`REFUSED: ${refusals.join('; ')}. No model was called and nothing was written.`);
    process.exit(2);
  }
  const ceilingInr = CEILINGS[stage][seat];
  const withheld = WITHHELD_SETS[withheldArg];
  const plan = stage === 1
    ? { stage: 1, wordings: ['w1', 'w2', 'w3', 'control'], repeats: 1 }
    : { stage: 2, wordings: [wordingArg], repeats: 2 };
  console.log(`SEAT ${seat} · model ${model} · stage ${plan.stage} · withheld ${withheld.length} (${withheld.join(', ')}) · ceiling Rs ${ceilingInr}`);
  const { rows, spent, stopped } = await runStage({ ...plan, seat, model, withheld, ceilingInr, fixtures, handbooks, log: (s) => console.log(s) });
  const tip = execFileSync('git', ['rev-parse', 'HEAD'], { cwd: ROOT }).toString().trim();
  const now = new Date().toISOString();
  const first = rows[0] || {};
  fs.mkdirSync(OUT_DIR, { recursive: true });
  const file = path.join(OUT_DIR, `p5b_hands_table_${seat}_s${plan.stage}_w${withheld.length}_${now.replace(/[-:]/g, '').replace(/\..*$/, '')}Z.md`);
  fs.writeFileSync(file, table(rows, { tip, now, fingerprint: fixtures.fingerprint, seat,
    victorModel: first.victor_model, donnaModel: first.donna_model, offeredCount: first.offered_count === undefined ? null : first.offered_count,
    stages: `stage ${plan.stage} (${plan.wordings.join(', ')}), ${plan.repeats} repeat(s)`, ceiling: ceilingInr, spent, stopped, withheld,
    moneyBlock: first.money_block, bookedBlock: first.booked_block,
    handbooks: handbooks.map((h) => `${h.field} (${(h.full_md || '').length} chars whole, ${(h.index_md || '').length} index)`).join(' · ') }));
  console.log(`TABLE ${path.relative(ROOT, file)} · seat ${seat} · ${rows.length} turns · Rs ${spent.toFixed(2)}${stopped ? ` · STOPPED: ${stopped}` : ''}`);
}

if (require.main === module) {
  main().catch((e) => { console.log(`RIG FAILED: ${(e && e.message) || String(e)}`); process.exit(1); });
}
